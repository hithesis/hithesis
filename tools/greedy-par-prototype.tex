  {
    \par
    \group_begin:

\cs_new_protected:Npn \__hg_run:
  {
    \tl_set:Nn \l__hg_line_tl { \tex_kern:D 2 \ccwd \scan_stop: }
    \dim_zero:N \l__hg_pool_dim
    \bool_set_true:N \l__hg_first_bool
    \seq_map_inline:Nn \l__hg_items_seq { \__hg_step:n {##1} }
    % 末行
    \tl_if_empty:NF \l__hg_line_tl { \exp_args:NV \__hg_emit_last:n \l__hg_line_tl }
  }

\cs_new_protected:Npn \__hg_step:n #1
  {
    % 候选 = 现行 + 新字
    \tl_set:Nx \l__hg_cand_tl { \exp_not:V \l__hg_line_tl \exp_not:n {#1} }
    \exp_args:NV \__hg_natural:n \l__hg_cand_tl
    \dim_compare:nNnTF { \box_wd:N \l__hg_trial_box } > { \l__hg_measure_dim }
      { \__hg_overflow:n {#1} }
      {
        \tl_set_eq:NN \l__hg_line_tl \l__hg_cand_tl
        \tl_if_in:NnT \c__hg_closers_tl {#1}
          { \dim_add:Nn \l__hg_pool_dim { 0.3 \ccwd } }
      }
  }

\cs_new_protected:Npn \__hg_overflow:n #1
  {
    \tl_if_in:NnTF \c__hg_closers_tl {#1}
      {
        % 禁则：先试拉回（定宽重排，预算=行内 0.3 格 caps）
        \dim_compare:nNnTF
          { \box_wd:N \l__hg_trial_box - \l__hg_measure_dim }
          > { \l__hg_pool_dim + 0.5 \ccwd }
          {
            % 悬挂（口袋加行末闭号自身半格都不够）
            \tl_set:Nx \l__hg_cand_tl
              {
                \exp_not:V \l__hg_line_tl \exp_not:n {#1}
                \exp_not:N \tex_kern:D \dim_eval:n { 0.25 bp - 1 em }
                \exp_not:N \scan_stop:
              }
            \exp_args:NV \__hg_emit:n \l__hg_cand_tl
            \tl_clear:N \l__hg_line_tl
            \dim_zero:N \l__hg_pool_dim
          }
          {
            \exp_args:NV \__hg_emit:n \l__hg_cand_tl
            \tl_clear:N \l__hg_line_tl
            \dim_zero:N \l__hg_pool_dim
          }
      }
      {
        \tl_if_in:NnTF \c__hg_openers_tl {#1}
          {
            % 开号推下行
            \exp_args:NV \__hg_emit:n \l__hg_line_tl
            \tl_set:Nn \l__hg_line_tl {#1}
            \dim_zero:N \l__hg_pool_dim
          }
          {
            % 普通字：口袋内塞字，超了推下行
            \dim_compare:nNnTF
              { \box_wd:N \l__hg_trial_box - \l__hg_measure_dim }
              > { \l__hg_pool_dim }
              {
                \exp_args:NV \__hg_emit:n \l__hg_line_tl
                \tl_set:Nn \l__hg_line_tl {#1}
              }
              {
                \exp_args:NV \__hg_emit:n \l__hg_cand_tl
                \tl_clear:N \l__hg_line_tl
              }
            \dim_zero:N \l__hg_pool_dim
          }
      }
  }
