\documentclass[stage=proposal,degree-level=doctor,campus=harbin]{hithesis}
% 开题报告与中期报告。写哪一种由 stage 决定：
%   stage=proposal   开题报告
%   stage=interim    中期报告
% 各用哪些文件写在正文里对应的 \hitif[stage=…] 里，不必改这一行以外的东西。
% 学位论文在 final.tex。
%
% 其余文档类选项的含义见手册，常用的几个：
% degree-level=doctor|master|bachelor|postdoc   必填
% degree-type=academic|professional             学术学位还是专业学位，默认 academic
% campus=harbin|shenzhen|weihai                 默认 harbin

\graphicspath{{figures/}}

% 示例宏包。装的是规范管不着、写论文常用的东西：\num 的分组空格、算法、
% 代码清单、数学记号速记等，见手册。要改就复制一份到自己的目录再改。
\usepackage{hithesis}

% 元信息（作者、题目、院系等）两份主文件共用一份。
\input{info}

\begin{document}
% 报告由哪些部分组成、按什么次序排，就是下面写的这些。封面、目录由类按模板判断排不排，
% 要改判断就在可选参数里写 shown=true 或 shown=false。正文第一个标题自动进正文，
% 参考文献自动进后置，\frontmatter、\mainmatter、\backmatter 不用写。

% 封面上的元信息（报告类型、日期……），开题与中期各一份。
\hitif[stage=proposal]{\input{proposal/frontmatter/cover}}{\input{interim/frontmatter/cover}}
\cover
\tableofcontents

% 正文。各校区、各学位的报告模板要填的项目不一样，按模板各写了一份，由条件挑：
% 逗号是“且”，分号是“或”，竖线并列几个取值，! 是“不是”。下面各行两两互斥，
% 一份报告只命中一行。用不着的行连同文件删掉即可。
% 本部研究生、威海博士：研究生院的开题报告模板
\hitif[stage=proposal,campus=harbin,degree-level=!bachelor;
       stage=proposal,campus=weihai,degree-level=doctor]{\input{proposal/mainmatter/graduate}}{}
% 威海硕士：威海校区的开题报告范例
\hitif[stage=proposal,campus=weihai,degree-level=master]{\input{proposal/mainmatter/weihai-master}}{}
% 本部本科、威海本科：教务处的开题报告模板，两个校区各一份
\hitif[stage=proposal,campus=harbin,degree-level=bachelor]{\input{proposal/mainmatter/bachelor}}{}
\hitif[stage=proposal,campus=weihai,degree-level=bachelor]{\input{proposal/mainmatter/weihai-bachelor}}{}
% 深圳研究生：深圳校区的开题报告表单，“主要参考文献”一节在表单里，\bibliography 也写在里面
\hitif[stage=proposal,campus=shenzhen,degree-level=!bachelor]{\input{proposal/mainmatter/shenzhen-graduate}}{}
% 深圳本科：深圳校区的开题报告模板，中英文各一份
\hitif[stage=proposal,campus=shenzhen,degree-level=bachelor,lang=zh]{\input{proposal/mainmatter/shenzhen-bachelor}}{}
\hitif[stage=proposal,campus=shenzhen,degree-level=bachelor,lang=en]{\input{proposal/mainmatter/shenzhen-bachelor-en}}{}
% 本部与威海研究生：研究生院的中期报告模板
\hitif[stage=interim,campus=harbin|weihai,degree-level=!bachelor]{\input{interim/mainmatter/graduate}}{}
% 本部本科：教务处的中期报告模板
\hitif[stage=interim,campus=harbin,degree-level=bachelor]{\input{interim/mainmatter/bachelor}}{}
% 威海本科：威海校区的中期报告模板
\hitif[stage=interim,campus=weihai,degree-level=bachelor]{\input{interim/mainmatter/weihai-bachelor}}{}
% 深圳研究生：深圳校区的中期报告表单，指导教师与检查小组意见也在正文里
\hitif[stage=interim,campus=shenzhen,degree-level=!bachelor]{\input{interim/mainmatter/shenzhen-graduate}}{}
% 深圳本科：深圳校区的中期报告模板，中英文各一份
\hitif[stage=interim,campus=shenzhen,degree-level=bachelor,lang=zh]{\input{interim/mainmatter/shenzhen-bachelor}}{}
\hitif[stage=interim,campus=shenzhen,degree-level=bachelor,lang=en]{\input{interim/mainmatter/shenzhen-bachelor-en}}{}

% 参考文献排在正文最后。深圳研究生的两份表单、本部本科与深圳本科的中期模板没有这一项。
\hitif[stage=proposal,campus=!shenzhen;
       stage=proposal,degree-level=bachelor;
       stage=interim,campus=weihai;
       stage=interim,campus=harbin,degree-level=!bachelor]{\bibliography{final/backmatter/reference}}{}
\end{document}
