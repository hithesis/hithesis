\documentclass[stage=proposal,degree-level=doctor,campus=harbin]{hithesis}
% 开题报告与中期报告。写哪一种由 stage 决定：
%   stage=proposal   开题报告
%   stage=interim    中期报告
% 各用哪些文件写在下面对应的 \hitsetup[stage=…] 里，不必改这一行以外的东西。
% 学位论文在 final.tex。
%
% 其余文档类选项的含义见手册，常用的几个：
% degree-level=doctor|master|bachelor|postdoc   必填
% degree-type=academic|professional             学术学位还是专业学位，默认 academic
% campus=harbin|shenzhen|weihai                 默认 harbin

\graphicspath{{figures/}}

% 示例宏包。装的是规范管不着、写论文常用的东西：\num 的分组空格、算法、
% 代码清单、数学记号速记等，见手册。要改就复制一份到自己的目录再改。
\usepackage{hithesis}

% 元信息（作者、题目、院系等）两份主文件共用一份。
\input{info}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% 两种报告的部件。方括号里是条件，只有当前这一份命中的那几组会生效。
% 顺序由类按规范决定，这里只说哪个部件用哪个文件；不写的部件不排。
%
% 各校区、各学位的报告模板要填的项目不一样，正文按模板各写了一份，由条件挑：
% 逗号是“且”，分号是“或”，竖线并列几个取值，! 是“不是”。写正文的那几组两两互斥，
% 一份报告只命中一组；封面那组每份都命中，只设封面，与正文那组不碰同一个部件，
% 所以先后无关。用不着的组连同文件删掉即可。
% 参考文献交给模板排在正文最后；深圳研究生的开题表单自己列了“主要参考文献”一节，
% 在正文里写 \bibliography，那两组不写 bibliography。
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\hitsetup[stage=proposal]{ struct = {
  frontmatter = { cover = {proposal/frontmatter/cover} },
} }
% 本部研究生、威海博士：研究生院的开题报告模板
\hitsetup[stage=proposal,campus=harbin,degree-level=!bachelor;
          stage=proposal,campus=weihai,degree-level=doctor]{ struct = {
  mainmatter = {proposal/mainmatter/graduate},
  backmatter = { bibliography = {final/backmatter/reference} },
} }
% 威海硕士：威海校区的开题报告范例
\hitsetup[stage=proposal,campus=weihai,degree-level=master]{ struct = {
  mainmatter = {proposal/mainmatter/weihai-master},
  backmatter = { bibliography = {final/backmatter/reference} },
} }
% 本部本科、威海本科：教务处的开题报告模板，两个校区各一份
\hitsetup[stage=proposal,campus=harbin,degree-level=bachelor]{ struct = {
  mainmatter = {proposal/mainmatter/bachelor},
  backmatter = { bibliography = {final/backmatter/reference} },
} }
\hitsetup[stage=proposal,campus=weihai,degree-level=bachelor]{ struct = {
  mainmatter = {proposal/mainmatter/weihai-bachelor},
  backmatter = { bibliography = {final/backmatter/reference} },
} }
% 深圳研究生：深圳校区的开题报告表单
\hitsetup[stage=proposal,campus=shenzhen,degree-level=!bachelor]{ struct = {
  mainmatter = {proposal/mainmatter/shenzhen-graduate},
} }
% 深圳本科：深圳校区的开题报告模板，中英文各一份
\hitsetup[stage=proposal,campus=shenzhen,degree-level=bachelor,lang=zh]{ struct = {
  mainmatter = {proposal/mainmatter/shenzhen-bachelor},
  backmatter = { bibliography = {final/backmatter/reference} },
} }
\hitsetup[stage=proposal,campus=shenzhen,degree-level=bachelor,lang=en]{ struct = {
  mainmatter = {proposal/mainmatter/shenzhen-bachelor-en},
  backmatter = { bibliography = {final/backmatter/reference} },
} }

\hitsetup[stage=interim]{ struct = {
  frontmatter = { cover = {interim/frontmatter/cover} },
} }
% 本部与威海研究生：研究生院的中期报告模板
\hitsetup[stage=interim,campus=harbin|weihai,degree-level=!bachelor]{ struct = {
  mainmatter = {interim/mainmatter/graduate},
  backmatter = { bibliography = {final/backmatter/reference} },
} }
% 本部本科：教务处的中期报告模板，没有参考文献一项
\hitsetup[stage=interim,campus=harbin,degree-level=bachelor]{ struct = {
  mainmatter = {interim/mainmatter/bachelor},
} }
% 威海本科：威海校区的中期报告模板
\hitsetup[stage=interim,campus=weihai,degree-level=bachelor]{ struct = {
  mainmatter = {interim/mainmatter/weihai-bachelor},
  backmatter = { bibliography = {final/backmatter/reference} },
} }
% 深圳研究生：深圳校区的中期报告表单，指导教师与检查小组意见也在正文里
\hitsetup[stage=interim,campus=shenzhen,degree-level=!bachelor]{ struct = {
  mainmatter = {interim/mainmatter/shenzhen-graduate},
} }
% 深圳本科：深圳校区的中期报告模板，中英文各一份，没有参考文献一项
\hitsetup[stage=interim,campus=shenzhen,degree-level=bachelor,lang=zh]{ struct = {
  mainmatter = {interim/mainmatter/shenzhen-bachelor},
} }
\hitsetup[stage=interim,campus=shenzhen,degree-level=bachelor,lang=en]{ struct = {
  mainmatter = {interim/mainmatter/shenzhen-bachelor-en},
} }

% 上面这些 \hitsetup 都写在导言区。走 biber 时这是必须的：biblatex 得在
% \begin{document} 之前装载，struct 里的 bibliography 也要赶在那之前换算成
% \addbibresource。走传统 BibTeX 时写在正文里也行，但没有理由分两处。
\begin{document}
% 前置、正文、后置由类按 structure 自动排，三个 matter 宏不用写。
% 想在两段之间插点别的东西时，把 \hitfrontmatter、\hitmainmatter、
% \hitbackmatter 显式写出来即可，写了的就在写的位置排。详见手册。
\end{document}
