\documentclass[kind=proposal,degree-level=doctor,campus=shenzhen]{hithesis}
% 开题报告与中期报告。写哪一种由 kind 决定：
%   kind=proposal   开题报告
%   kind=interim    中期报告
% 各用哪些文件写在下面对应的 \hitsetup[kind=…] 里，不必改这一行以外的东西。
% 学位论文在 thesis.tex。
%
% 其余文档类选项的含义见手册，常用的几个：
% degree-level=doctor|master|bachelor|postdoc   必填
% degree-type=academic|professional             学术学位还是专业学位，默认 academic
% campus=harbin|shenzhen|weihai                 默认 harbin

\graphicspath{{figures/}}

% 示例宏包。里面是示例文档自己要用的东西：术语词条、染色体示意图的绘图宏、
% 数学记号速记、\num 的分组空格。都不是规范要求的，复制一份改成自己的。
\usepackage{hithesis}

% 元信息（作者、题目、院系等）两份主文件共用一份。
\input{info}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% 两种报告的部件。方括号里是条件，只有当前这一种那一组会生效。
% 顺序由类按规范决定，这里只说哪个部件用哪个文件；不写的部件不排。
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% 正文按校区与学位分了几份，示例排的是深圳博士那份。换成自己那一份即可：
%   report/proposal/mainmatter/shenzhen_doctor    深圳博士
%   report/proposal/mainmatter/harbin_bachelor    本部本科
%   report/proposal/mainmatter/weihai_bachelor    威海本科
\hitsetup[kind=proposal]{ struct = {
  frontmatter = { cover = {report/proposal/frontmatter/cover} },
  mainmatter  = {report/proposal/mainmatter/shenzhen_doctor},
} }

%   report/interim/mainmatter/shenzhen_doctor     深圳博士
%   report/interim/mainmatter/harbin_master       本部硕士
%   report/interim/mainmatter/weihai_bachelor     威海本科
\hitsetup[kind=interim]{ struct = {
  frontmatter = { cover = {report/interim/frontmatter/cover} },
  mainmatter  = {report/interim/mainmatter/shenzhen_doctor},
} }

% 上面这些 \hitsetup 都写在导言区。走 biber 时这是必须的：biblatex 得在
% \begin{document} 之前装载，struct 里的 bibliography 也要赶在那之前换算成
% \addbibresource。走传统 BibTeX 时写在正文里也行，但没有理由分两处。
\begin{document}
% 前置、正文、后置由类按 structure 自动排，三个 matter 宏不用写。
% 想在两段之间插点别的东西时，把 \hitfrontmatter、\hitmainmatter、
% \hitbackmatter 显式写出来即可，写了的就在写的位置排。详见手册。
\end{document}
