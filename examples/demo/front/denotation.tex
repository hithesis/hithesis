\begin{denotation}
\item[$\mathbb{R}$] 实数集
\item[$\mathbb{N}$] 非负整数集
\item[Hz] 频率单位赫[兹]，$\mathrm{s}^{-1}$
\item[N] 力的单位牛[顿]，$\mathrm{kg}\cdot\mathrm{m}/\mathrm{s}^{2}$
\item[Pa] 压力单位帕[斯卡]，$\mathrm{N}/\mathrm{m}^{2}$
\item[J] 能量单位焦[耳]，$\mathrm{N}\cdot\mathrm{m}$
\item[W] 功率单位瓦[特]，$\mathrm{J}/\mathrm{s}$
\end{denotation}
