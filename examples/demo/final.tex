\usepackage{hithesis}

% 元信息（作者、题目、院系等）两份主文件共用一份。

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% 两版的部件。方括号里是条件，只有当前这一版那一组会生效。
% 顺序由类按规范决定，这里只说哪个部件用哪个文件；不写的部件不排。
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% 两组条件互斥，任何一份文档只命中一组，所以先后无关，各组必须自足。
% （\hitsetup 是后写覆盖先写，条件只决定这一句做不做、不决定谁优先。写成互斥的
% 就不必琢磨顺序；写成「一组打底、另一组只改几项」也行，但那样顺序就有讲究了。）
\hitsetup[lang=zh]{ struct = {
  frontmatter = {
    cover      = {final/frontmatter/cover},
    abstract   = {final/frontmatter/abstract},
    list-of-symbols = {final/frontmatter/symbols},
    list-of-abbreviations = true,
    table-of-contents = true,
  },
  % 正文各章按写的顺序排，结论是正文的最后一章。
  mainmatter = {final/mainmatter/chapter-1, final/mainmatter/chapter-2,
                final/mainmatter/chapter-3, final/mainmatter/conclusion},
  backmatter = {
    bibliography     = {final/backmatter/reference},
    appendix         = {final/backmatter/appA},
    publications     = {final/backmatter/publications},
    defense          = auto,
    index            = {final/backmatter/ceindex},
    declarations     = true,
    acknowledgements = {final/backmatter/acknowledgements},
    resume           = {final/backmatter/resume},
  },
} }
\hitsetup{ bib = { bibtex = { style = {hithesis} } } }   % 换 .bst 改这里，标点全半角走 bibpunct 旋钮

% 英文版：致谢在前置，没有符号表、附录、索引、简历。正文与结论都在 english 那一个文件里。
\hitsetup[lang=en]{ struct = {
  frontmatter = {
    cover             = {final/frontmatter/cover-en},
    abstract          = {final/frontmatter/abstract},
    table-of-contents = true,
    % 图索引与表索引不写就是 auto：规范只要求英文版排，模板按 lang 自己定。
    % 想在中文版也排就写 list-of-figures = true，想在英文版关掉就写 false。
    acknowledgements  = {final/frontmatter/acknowledgements-en},
  },
  mainmatter = {final/mainmatter/english},
  backmatter = {
    bibliography  = {final/backmatter/reference},
    publications  = {final/backmatter/publications},
    declarations = true,
  },
} }

% 上面这些 \hitsetup 都写在导言区。走 biber 时这是必须的：biblatex 得在
% \begin{document} 之前装载，struct 里的 bibliography 也要赶在那之前换算成
% \addbibresource。走传统 BibTeX 时写在正文里也行，但没有理由分两处。
\begin{document}
% 前置、正文、后置由类按 structure 自动排，三个 matter 宏不用写。
% 想在两段之间插点别的东西时，把 \hitfrontmatter、\hitmainmatter、
% \hitbackmatter 显式写出来即可，写了的就在写的位置排。详见手册。
\end{document}
